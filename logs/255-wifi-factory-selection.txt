#!/system/bin/sh

gpio_num=$1
gpio_state=$2
((group_num = gpio_num / 8))
((port_num = gpio_num % 8))
echo current modify GPIO${group_num}_${port_num}

gpio_group_dir_addr_arr=("0xfffff000" "0xfffff000" "0xf8b22400" "0xf8b23400" "0xf8b24400" "0xf8004400"  "0xf8b26400" "0xf8b27400" "0xf8b28400")

function print_help() {
    echo "$0 [gpio_num] [on|off]"
    echo "e.g."
    echo "    $0 44 on"
    echo "    $0 44 off"
}

if [ "$#" != "2" ]; then
    echo "wrong args!"
    print_help
    exit 1
elif [ -z "$gpio_state" ]; then
    echo "wrong args!"
    print_help
    exit 1
fi

if [ -n "$gpio_num" ] && [ "$gpio_num" -eq "$gpio_num" ]; then

    if [ "$group_num" -gt "8" ] ; then
        echo "group_num > 8 is not support"
        print_help
        exit 1
    elif [ "$group_num" -lt "2" ] ; then
        echo "group_num < 2 is not support"
        print_help
        exit 1
    fi

    gpio_dir_addr=${gpio_group_dir_addr_arr[$group_num]}
    cur_output_state=$(busybox devmem ${gpio_dir_addr} 32)
    #echo $cur_output_state
    gpio_left_shift=$((1 << $port_num))
    target_output_state=$((cur_output_state | gpio_left_shift))
    target_output_state=$(busybox printf "0x%04x" $target_output_state)
    #echo $target_output_state
    busybox devmem ${gpio_dir_addr} 32 ${target_output_state}
    current_port_reg=$((4 << $port_num))
    gpio_value_addr=${gpio_dir_addr:0:7}$(busybox printf "%03x" $current_port_reg)
    #echo $gpio_value_addr
    cur_output_value=$(busybox devmem ${gpio_value_addr} 32)
    #echo $cur_output_value
    
    gpio_high_value=$(busybox printf "0x%02x" $gpio_left_shift)
    gpio_low_value=0x0
    if [ "$gpio_state" == "on" ]; then
        busybox devmem ${gpio_value_addr} 32 $gpio_high_value
    else
        busybox devmem ${gpio_value_addr} 32 $gpio_low_value
    fi
else
    echo "wrong args!"
    print_help
