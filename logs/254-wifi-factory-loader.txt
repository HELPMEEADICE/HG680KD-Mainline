#!/system/bin/sh

BOARDID=`getprop ro.fh.boardid`
LSADCIX=`getprop ro.fh.lsadcix`
if [ "$BOARDID" == "1" ] || [ "$BOARDID" == "2" ] || [ "$BOARDID" == "4" ];then
    #for lan & pon rtl8761b
    ETHERNETMAC=`getprop ro.mac`
    if [ -n "$ETHERNETMAC" ]; then
        SimpleMac=`echo "$ETHERNETMAC" | busybox sed 's/://g'`
        #bt mac = eth mac + 1
        LowBtMac=$((16#${SimpleMac:6:6}+1))
        HightBtMac=$((16#${SimpleMac:0:6}))
        if [ $LowBtMac -gt $((16#FFFFFF)) ];then
            LowBtMac=$(($LowBtMac-$((16#FFFFFF))-1))
            HightBtMac=$(($HightBtMac+1))
        fi
        BtMac=`busybox printf "%06x%06x" $HightBtMac $LowBtMac | busybox tr a-z A-Z`
        FinalBtMac=$(echo "${BtMac:0:2}:${BtMac:2:2}:${BtMac:4:2}:${BtMac:6:2}:${BtMac:8:2}:${BtMac:10:2}")
        BTMACSTATUS=`getprop persist.vendor.rtkbt.bdaddr_path`
        if [ "$BTMACSTATUS" != "default" ];then
            setprop persist.vendor.rtkbt.bdaddr_path default
        fi
        setprop ro.boot.btmacaddr $FinalBtMac
    fi
elif [ "$BOARDID" == "0" ] ;then
    insmod /vendor/lib/modules/hi_sdio_detect.ko
    insmod /vendor/lib/modules/btmtksdio.ko
    insmod /vendor/lib/modules/wlan_mt7663_sdio_prealloc.ko
    insmod /vendor/lib/modules/uwe5621_bsp_sdio.ko
    insmod /vendor/lib/modules/uwe5621_bt_sdio.ko
    chown bluetooth net_bt_admin /dev/ttyBT0
    chown bluetooth net_bt_admin /sys/bus/platform/devices/mtty.0/at
    chown bluetooth net_bt_admin /sys/bus/platform/devices/mtty.0/woble_set
    chmod 660 /dev/ttyBT0
    chmod 660 /sys/bus/platform/devices/mtty.0/at
    chmod 660 /sys/bus/platform/devices/mtty.0/woble_set
fi

if [ "$BOARDID" == "7" ] ;then
    if [ "$LSADCIX" == "12" ] || [ "$LSADCIX" == "8" ];then
        #reset only in MT7920QEN
        chmod 644 /sys/hi_rfkill/reset_onresume
        echo 0 > /sys/hi_rfkill/reset_onresume
    else
        setprop persist.wifi.ap.interface ap0
        echo 0 > /proc/sys/kernel/printk
        /system/bin/opt/etc/gpio_ctl.sh 35 off
        sleep 0.2                
        /system/bin/opt/etc/gpio_ctl.sh 35 on
        sleep 0.5                         
        echo 1 > /sys/class/pci_bus/0000\:01/device/remove
        echo 1 > /sys/class/pci_bus/0000\:00/rescan
        insmod /vendor/lib/modules/btmtk_uart_unify.ko
        start mtk_uart_launcher
    fi
else
    #reset only in MT7920QEN
    chmod 644 /sys/hi_rfkill/reset_onresume
    echo 0 > /sys/hi_rfkill/reset_onresume
fi

setprop ctl.start wifi_bt

#bt wifi check
/system/bin/opt/etc/hw.detect.sh &
